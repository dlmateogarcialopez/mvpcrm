@app.route('/webhook', methods=['GET'])
def webhook_whatsapp():
    try:

        VERIFY_TOKEN = os.getenv("VERIFY_TOKEN")

        # Parsear parámetros de la solicitud de verificación
        mode = request.args.get('hub.mode')
        token = request.args.get('hub.verify_token')
        challenge = request.args.get('hub.challenge')

        # Verificar si se enviaron mode y token
        if mode and token:
            if mode == 'subscribe' and token == VERIFY_TOKEN:
                print("WEBHOOK_VERIFIED")
                return challenge, 200  # Retorna el challenge y código 200
            else:
                return '', 403  # Token no coincide, retorna 403 Forbidden
        else:
            return '', 400  # Faltan parámetros, retorna 400 Bad Request

    except Exception as e:
        # Manejo de errores (equivalente a boom.badImplementation)
        return jsonify({"error": "webhook token error", "details": str(e)}), 500
